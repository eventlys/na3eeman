import 'package:flutter/material.dart';

class CustomAnimatedIcon extends StatefulWidget {
  final IconData icon;
  final VoidCallback onTap;
  final Color backgroundColor;
  final Color iconColor;
  final double size;
  final Duration duration;

  const CustomAnimatedIcon({
    super.key,
    required this.icon,
    required this.onTap,
    this.backgroundColor = const Color(0xFF4CAF50),
    this.iconColor = Colors.white,
    this.size = 20.0,
    this.duration = const Duration(milliseconds: 1000),
  });

  @override
  State<CustomAnimatedIcon> createState() => _CustomAnimatedIconState();
}

class _CustomAnimatedIconState extends State<CustomAnimatedIcon>
    with SingleTickerProviderStateMixin {
  late AnimationController _controller;
  late Animation<double> _animation;

  @override
  void initState() {
    super.initState();
    _controller = AnimationController(
      vsync: this,
      duration: widget.duration,
    );

    _animation = TweenSequence<double>(
      [
        TweenSequenceItem(tween: Tween(begin: 1.0, end: 1.3), weight: 50),
        TweenSequenceItem(tween: Tween(begin: 1.3, end: 1.0), weight: 50),
      ],
    ).animate(CurvedAnimation(
      parent: _controller,
      curve: Curves.elasticOut,
    ));
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  void _handleTap() {
    _controller.forward(from: 0);
    widget.onTap();
  }

  @override
  Widget build(BuildContext context) {
    return ScaleTransition(
      scale: _animation,
      child: GestureDetector(
        onTap: _handleTap,
        child: Container(
          padding: const EdgeInsets.all(8),
          decoration: BoxDecoration(
            color: widget.backgroundColor,
            borderRadius: BorderRadius.circular(8),
          ),
          child: Icon(
            widget.icon,
            color: widget.iconColor,
            size: widget.size,
          ),
        ),
      ),
    );
  }
}

class ThemeManager with ChangeNotifier {
  ThemeMode _themeMode = ThemeMode.system;
  bool _highContrast = false;
  UserGender _userGender = UserGender.male;

  ThemeMode get themeMode => _themeMode;
  bool get highContrast => _highContrast;
  UserGender get userGender => _userGender;

  void setThemeMode(ThemeMode mode) {
    _themeMode = mode;
    notifyListeners();
  }

  void toggleHighContrast() {
    _highContrast = !_highContrast;
    notifyListeners();
  }

  void setUserGender(UserGender gender) {
    _userGender = gender;
    notifyListeners();
  }

  ThemeData getCurrentTheme(BuildContext context) {
    final brightness = MediaQuery.of(context).platformBrightness;
    bool isDark;

    switch (_themeMode) {
      case ThemeMode.system:
        isDark = brightness == Brightness.dark;
        break;
      case ThemeMode.dark:
        isDark = true;
        break;
      case ThemeMode.light:
        isDark = false;
        break;
    }

    if (_highContrast) {
      return AppThemes.getHighContrastTheme(isDark);
    }

    // Use gender-specific theme if gender is set
    return AppThemes.getGenderTheme(_userGender, isDark);
  }
}

enum ButtonVariant { filled, outlined, text }

class PrimaryButton extends StatelessWidget {
  final String text;
  final VoidCallback onPressed;
  final bool showLeftIcon;
  final bool showRightIcon;
  final IconData? leftIcon;
  final IconData? rightIcon;
  final Color buttonColor;
  final Color textColor;
  final ButtonVariant variant;
  final bool isLoading;

  const PrimaryButton({
    super.key,
    required this.text,
    required this.onPressed,
    this.showLeftIcon = false,
    this.showRightIcon = false,
    this.leftIcon,
    this.rightIcon,
    this.buttonColor = AppColor.primary,
    this.textColor = AppColor.light,
    this.variant = ButtonVariant.filled,
    this.isLoading = false,
  });

  @override
  Widget build(BuildContext context) {
    return ElevatedButton(
      onPressed: isLoading ? null : onPressed,
      style: _getButtonStyle(context),
      child: isLoading
          ? const SizedBox(
              width: 20,
              height: 20,
              child: CircularProgressIndicator(
                strokeWidth: 2,
                valueColor: AlwaysStoppedAnimation<Color>(AppColor.light),
              ),
            )
          : Row(
              mainAxisSize: MainAxisSize.min,
              children: [
                if (showLeftIcon && leftIcon != null)
                  Icon(leftIcon, size: 20, color: _getIconColor()),
                if (showLeftIcon && leftIcon != null)
                  const SizedBox(width: 8),
                Text(
                  text,
                  style: TextStyle(
                    color: _getTextColor(),
                    fontWeight: FontWeight.w600,
                  ),
                ),
                if (showRightIcon && rightIcon != null)
                  const SizedBox(width: 8),
                if (showRightIcon && rightIcon != null)
                  Icon(rightIcon, size: 20, color: _getIconColor()),
              ],
            ),
    );
  }

  ButtonStyle _getButtonStyle(BuildContext context) {
    switch (variant) {
      case ButtonVariant.filled:
        return ElevatedButton.styleFrom(
          backgroundColor: buttonColor,
          foregroundColor: textColor,
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(8),
          ),
        );
      case ButtonVariant.outlined:
        return ElevatedButton.styleFrom(
          backgroundColor: AppColor.trans,
          foregroundColor: buttonColor,
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(8),
            side: BorderSide(color: buttonColor, width: 2),
          ),
        );
      case ButtonVariant.text:
        return ElevatedButton.styleFrom(
          backgroundColor: AppColor.trans,
          foregroundColor: buttonColor,
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(8),
          ),
          elevation: 0,
        );
    }
  }

  Color _getTextColor() {
    switch (variant) {
      case ButtonVariant.filled:
        return textColor;
      case ButtonVariant.outlined:
      case ButtonVariant.text:
        return buttonColor;
    }
  }

  Color _getIconColor() {
    switch (variant) {
      case ButtonVariant.filled:
        return textColor;
      case ButtonVariant.outlined:
      case ButtonVariant.text:
        return buttonColor;
    }
  }
}

class CustomNeumorphicIcon extends StatelessWidget {
  final IconData icon;
  final double size;
  final Color iconColor;
  final double depth;

  const CustomNeumorphicIcon({
    super.key,
    required this.icon,
    required this.size,
    required this.iconColor,
    this.depth = 2,
  });

  @override
  Widget build(BuildContext context) {
    return Container(
      width: size + 8,
      height: size + 8,
      decoration: BoxDecoration(
        color: AppColor.lightCard.withValues(alpha: 0.70),
        shape: BoxShape.circle,
        boxShadow: [
          // Dark shadow
          BoxShadow(
            color: AppColor.dark.withValues(alpha: 0.2),
            blurRadius: depth * 2,
            offset: Offset(depth, depth),
          ),
          // Light shadow
          BoxShadow(
            color: AppColor.light.withValues(alpha: 0.8),
            blurRadius: depth * 2,
            offset: Offset(-depth, -depth),
          ),
        ],
      ),
      child: Center(
        child: Icon(
          icon,
          size: size,
          color: iconColor,
        ),
      ),
    );
  }
}

  const CustomNeumorphicContainer({

    this.width = double.infinity,
    this.height = double.infinity,
    this.borderRadius = BorderRadius.zero,
    this.depth = 3,

  @override
  Widget build(BuildContext context) {
    return Container(
      width: width,
      height: height,
      decoration: BoxDecoration(
        borderRadius: borderRadius,
        color: AppColor.lightCard.withValues(alpha: 0.70),
        boxShadow: [
          BoxShadow(
            color: AppColor.primary.withValues(alpha: 0.4), // Approximating 0x6699806C which is #806C99 with 40% opacity
            blurRadius: depth * 2,
            offset: Offset(depth, depth),
            spreadRadius: 0,
          ),
        ],
      ),
      child: child,
    );
  }
}

// For more customizable inner shadows
class CustomInnerShadowPainter extends CustomPainter {
  final List<ShadowConfig> shadows;
  final double borderRadius;

  CustomInnerShadowPainter({
    required this.shadows,
    this.borderRadius = 0,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final rect = Rect.fromLTWH(0, 0, size.width, size.height);
    final radius = Radius.circular(borderRadius);

    for (final shadow in shadows) {
      final paint = Paint()
        ..color = shadow.color
        ..maskFilter = MaskFilter.blur(
          BlurStyle.normal,
          shadow.blurRadius,
        );

      canvas.drawRRect(
        RRect.fromRectAndRadius(rect, radius).deflate(shadow.spread),
        paint,
      );
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class ShadowConfig {
  final Color color;
  final double blurRadius;
  final double spread;

  const ShadowConfig({
    required this.color,
    required this.blurRadius,
    this.spread = 0,
  });
}

// For outer shadows with custom shapes
class CustomOuterShadowPainter extends CustomPainter {
  final List<BoxShadow> shadows;
  final double borderRadius;

  const CustomOuterShadowPainter({
    required this.shadows,
    required this.borderRadius,
  });

  @override
  void paint(Canvas canvas, Size size) {
    final rect = Rect.fromLTWH(0, 0, size.width, size.height);
    final rrect = RRect.fromRectAndRadius(rect, Radius.circular(borderRadius));

    for (final shadow in shadows) {
      final paint = Paint()
        ..color = shadow.color
        ..maskFilter = MaskFilter.blur(
          BlurStyle.normal,
          shadow.blurRadius,
        );

      canvas.drawRRect(
        rrect.shift(shadow.offset),
        paint,
      );
    }
  }

  @override
  bool shouldRepaint(covariant CustomPainter oldDelegate) => false;
}

class GlassSliverAppBar extends StatefulWidget {

  final bool floating;
  final bool snap;
  final double expandedHeight;
  final double collapsedHeight;
  final Widget? leading;
  final UserGender? userGender;
  final bool isDarkMode;

  const GlassSliverAppBar({
    super.key,
    required this.title,
    this.showBackButton = true,
    this.backgroundColor = AppColor.trans,
    this.textColor = AppColor.light,
    this.gradientColors,
    this.onBackPressed,
    this.actions,
    this.pinned = true,
    this.floating = false,
    this.snap = false,
    this.expandedHeight = 200,
    this.collapsedHeight = kToolbarHeight,
    this.leading,
    this.userGender,
    this.isDarkMode = false,
  });

  @override
  State<GlassSliverAppBar> createState() => _GlassSliverAppBarState();
}

class _GlassSliverAppBarState extends State<GlassSliverAppBar> {

  @override
  Widget build(BuildContext context) {
    final gradientColors = widget.gradientColors ?? _getGenderGradientColors();

    return SliverAppBar(
      centerTitle: true,
      backgroundColor: AppColor.trans,
      elevation: 0,
      leading: widget.leading ?? (widget.showBackButton ? _buildBackButton() : null),
      actions: widget.actions,
      pinned: widget.pinned,
      floating: widget.floating,
      snap: widget.snap,
      expandedHeight: widget.expandedHeight,
      collapsedHeight: widget.collapsedHeight,
      flexibleSpace: _buildFlexibleSpace(gradientColors),
    );
  }

  Widget _buildBackButton() {
    return IconButton(
      icon: Icon(Icons.arrow_back, color: _getIconColor()),
      onPressed: widget.onBackPressed ?? () => Navigator.of(context).pop(),
    );
  }

  Widget _buildFlexibleSpace(List<Color> gradientColors) {
    return Container(
      decoration: BoxDecoration(
        gradient: LinearGradient(
          colors: [
            gradientColors[0],
            gradientColors.length > 1 ? gradientColors[1] : gradientColors[0],
          ],
          begin: Alignment.topCenter,
          end: Alignment.bottomCenter,
        ),
      ),
      child: ClipRect(
        child: BackdropFilter(
          filter: ImageFilter.blur(sigmaX: 10, sigmaY: 10),
          child: Container(
            color: AppColor.trans,
            child: Align(
              alignment: Alignment.bottomCenter,
              child: Padding(
                padding: const EdgeInsets.all(16.0),
                child: Text(
                  widget.title,
                  style: TextStyle(
                    fontSize: 32,
                    fontWeight: FontWeight.bold,
                    color: _getTextColor(),
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }

  List<Color> _getGenderGradientColors() {
    if (widget.userGender != null) {
      switch (widget.userGender) {
        case UserGender.male:
          return widget.isDarkMode
              ? [AppColor.darkMaleBG.withValues(alpha: 0.3), AppColor.darkMaleBackground.withValues(alpha: 0.6)]
              : [AppColor.lightMaleBG.withValues(alpha: 0.3), AppColor.maleBackground.withValues(alpha: 0.6)];
        case UserGender.female:
          return widget.isDarkMode
              ? [AppColor.darkFemaleBG.withValues(alpha: 0.3), AppColor.darkFemaleBackground.withValues(alpha: 0.6)]
              : [AppColor.lightFemaleBG.withValues(alpha: 0.3), AppColor.femaleBackground.withValues(alpha: 0.6)];
        case UserGender.nonBinary:
          return widget.isDarkMode
              ? [AppColor.darkNonBinaryBG.withValues(alpha: 0.3), AppColor.darkNonBinaryBackground.withValues(alpha: 0.6)]
              : [AppColor.lightNonBinaryBG.withValues(alpha: 0.3), AppColor.nonBinaryBackground.withValues(alpha: 0.6)];
        default:
          return _getDefaultGradientColors();
      }
    }
    return _getDefaultGradientColors();
  }

  List<Color> _getDefaultGradientColors() {
    return widget.isDarkMode
        ? [AppColor.darkPrimary.withValues(alpha: 0.3), AppColor.darkSecondary.withValues(alpha: 0.6)]
        : [AppColor.lightPrimary.withValues(alpha: 0.3), AppColor.lightSecondary.withValues(alpha: 0.6)];
  }

  Color _getIconColor() {
    if (widget.userGender != null) {
      switch (widget.userGender) {
        case UserGender.male:
          return widget.isDarkMode ? AppColor.maleAccent : AppColor.maleAccent;
        case UserGender.female:
          return widget.isDarkMode ? AppColor.femaleAccent : AppColor.femaleAccent;
        case UserGender.nonBinary:
          return widget.isDarkMode ? AppColor.nonBinaryAccent : AppColor.nonBinaryAccent;
        default:
          return widget.isDarkMode ? AppColor.darkText : AppColor.lightText;
      }
    }
    return widget.isDarkMode ? AppColor.darkText : AppColor.lightText;
  }

  Color _getTextColor() {
    if (widget.userGender != null) {
      switch (widget.userGender) {
        case UserGender.male:
          return widget.isDarkMode ? AppColor.maleAccent : AppColor.maleAccent;
        case UserGender.female:
          return widget.isDarkMode ? AppColor.femaleAccent : AppColor.femaleAccent;
        case UserGender.nonBinary:
          return widget.isDarkMode ? AppColor.nonBinaryAccent : AppColor.nonBinaryAccent;
        default:
          return widget.isDarkMode ? AppColor.darkText : AppColor.lightText;
      }
    }
    return widget.textColor;
  }
}
import 'package:flutter/material.dart';

import '../constants/app_color.dart';

class NeomorphicSphere extends StatefulWidget {
  final String activeIconPath;
  final String inactiveIconPath;
  final double size;
  final Color sphereColor;
  final Color shadowColor;
  final VoidCallback? onTap;
  final String? navigationRoute;
  final Map<String, dynamic>? navigationArguments;
  final double iconSize;
  final bool isActive;
  final ValueChanged<bool>? onActiveChanged; // Callback when active state changes

  const NeomorphicSphere({
    super.key,
    required this.activeIconPath,
    required this.inactiveIconPath,
    this.size = 70,
    this.sphereColor = AppColor.lightSecondary, // Replaced Color(0xFFDFA8A4)
    this.shadowColor = AppColor.dark, // Replaced Color(0x80242424)
    this.onTap,
    this.navigationRoute,
    this.navigationArguments,
    this.iconSize = 30,
    this.isActive = false,
    this.onActiveChanged,
  });

  @override
  NeomorphicSphereState createState() => NeomorphicSphereState();
}

class NeomorphicSphereState extends State<NeomorphicSphere> with SingleTickerProviderStateMixin {
  bool _isPressed = false;
  late AnimationController _animationController;
  late Animation<Offset> _slideAnimation;
  late Animation<double> _opacityAnimation;

  @override
  void initState() {
    super.initState();

    _animationController = AnimationController(
      duration: const Duration(milliseconds: 300),
      vsync: this,
    );

    _slideAnimation = Tween<Offset>(
      begin: const Offset(0, 1.0), // Slide up from bottom
      end: Offset.zero,
    ).animate(CurvedAnimation(
      parent: _animationController,
      curve: Curves.easeOutCubic,
    ));

    _opacityAnimation = Tween<double>(
      begin: 0.0,
      end: 1.0,
    ).animate(CurvedAnimation(
      parent: _animationController,
      curve: Curves.easeInOut,
    ));

    // Start animation if initially active
    if (widget.isActive) {
      _animationController.forward();
    }
  }

  @override
  void didUpdateWidget(NeomorphicSphere oldWidget) {
    super.didUpdateWidget(oldWidget);

    // Animate when active state changes
    if (widget.isActive && !oldWidget.isActive) {
      _animationController.forward();
    } else if (!widget.isActive && oldWidget.isActive) {
      _animationController.reverse();
    }
  }

  @override
  void dispose() {
    _animationController.dispose();
    super.dispose();
  }

  void _handleTap(BuildContext context) {
    if (widget.isActive) {
      // If already active, just show press feedback and refresh/redirect
      setState(() => _isPressed = true);
      Future.delayed(const Duration(milliseconds: 150), () {
        setState(() => _isPressed = false);
      });

      // Execute custom onTap callback if provided
      widget.onTap?.call();

      // Refresh current page or navigate to same route
      if (widget.navigationRoute != null) {
        Navigator.pushReplacementNamed(
          context,
          widget.navigationRoute!,
          arguments: widget.navigationArguments,
        );
      }
    } else {
      // If not active, become active and navigate
      widget.onActiveChanged?.call(true);

      // Execute custom onTap callback if provided
      widget.onTap?.call();

      // Navigate to new page if route provided
      if (widget.navigationRoute != null) {
        if (widget.navigationArguments != null) {
          Navigator.pushNamed(
            context,
            widget.navigationRoute!,
            arguments: widget.navigationArguments,
          );
        } else {
          Navigator.pushNamed(context, widget.navigationRoute!);
        }
      }
    }
  }

  @override
  Widget build(BuildContext context) {
    final isActive = widget.isActive;
    final currentIconPath = isActive ? widget.activeIconPath : widget.inactiveIconPath;
    final currentIconSize = isActive ? widget.iconSize + 20 : widget.iconSize; // +20 when active

    return GestureDetector(
      onTapDown: (_) => setState(() => _isPressed = true),
      onTapUp: (_) => setState(() => _isPressed = false),
      onTapCancel: () => setState(() => _isPressed = false),
      onTap: () => _handleTap(context),
      child: Stack(
        alignment: Alignment.center,
        children: [
          // Neomorphic Sphere (only visible when active)
          if (isActive) ...[
            AnimatedBuilder(
              animation: _animationController,
              builder: (context, child) {
                return Transform.translate(
                  offset: _slideAnimation.value,
                  child: Opacity(
                    opacity: _opacityAnimation.value,
                    child: child,
                  ),
                );
              },
              child: Container(
                width: widget.size,
                height: widget.size,
                decoration: ShapeDecoration(
                  color: widget.sphereColor,
                  shape: const OvalBorder(),
                  shadows: _isPressed
                      ? [
                    // Inset shadow effect when pressed
                    BoxShadow(
                      color: widget.shadowColor.withValues(alpha: 0.2), // Replaced withOpacity
                      blurRadius: 5,
                      offset: const Offset(2, 2),
                      spreadRadius: -2,
                    ),
                  ]
                      : [
                    // Outer shadow for neomorphic effect
                    BoxShadow(
                      color: widget.shadowColor,
                      blurRadius: 15,
                      offset: const Offset(4, 4),
                      spreadRadius: 1,
                    ),
                  ],
                ),
              ),
            ),
          ],

          // Icon (always visible)
          Image.asset(
            currentIconPath,
            width: currentIconSize,
            height: currentIconSize,
            fit: BoxFit.contain,
            // No color filter applied - original colors
          ),
        ],
      ),
    );
  }
}
